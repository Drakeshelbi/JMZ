Información Básica del Servidor

-- Versión de SQL Server SELECT @@VERSION AS 'SQL Server Version'; -- Información del servidor SELECT SERVERPROPERTY('MachineName') AS 'Machine Name', SERVERPROPERTY('ServerName') AS 'Server Name', SERVERPROPERTY('InstanceName') AS 'Instance', SERVERPROPERTY('Edition') AS 'Edition', SERVERPROPERTY('ProductLevel') AS 'Product Level', SERVERPROPERTY('ProductVersion') AS 'Product Version'; -- Uso de CPU y memoria SELECT cpu_count AS 'Logical CPU Count', hyperthread_ratio AS 'Hyperthread Ratio', physical_memory_kb/1024 AS 'Physical Memory (MB)', sqlserver_start_time AS 'SQL Server Start Time' FROM sys.dm_os_sys_info;
Nota: Estas consultas proporcionan información fundamental sobre el entorno SQL Server al que estás conectado.

Información de Bases de Datos

-- Lista todas las bases de datos con información clave SELECT name AS 'Database Name', recovery_model_desc AS 'Recovery Model', state_desc AS 'State', compatibility_level AS 'Compatibility Level', user_access_desc AS 'User Access', size/128.0 AS 'Size (MB)', create_date AS 'Creation Date' FROM sys.databases ORDER BY name; -- Tamaño de archivos por base de datos SELECT DB_NAME(database_id) AS 'Database', name AS 'Logical Name', type_desc AS 'File Type', size/128.0 AS 'Size (MB)', max_size/128.0 AS 'Max Size (MB)', growth AS 'Growth (MB)', physical_name AS 'Physical Path' FROM sys.master_files ORDER BY DB_NAME(database_id), type_desc;
Importante: Verifica especialmente las bases de datos con recovery model FULL que no tengan backups recientes.

Configuraciones del Servidor

-- Configuraciones importantes del servidor SELECT name AS 'Config Name', value AS 'Value', value_in_use AS 'Value In Use', description AS 'Description' FROM sys.configurations WHERE name IN ( 'max server memory (MB)', 'min server memory (MB)', 'cost threshold for parallelism', 'max degree of parallelism', 'optimize for ad hoc workloads', 'backup compression default' ) ORDER BY name; -- Configuraciones de base de datos por defecto SELECT name AS 'Database', is_auto_close_on, is_auto_shrink_on, is_auto_create_stats_on, is_auto_update_stats_on, is_auto_update_stats_async_on, is_parameterization_forced FROM sys.databases ORDER BY name;
Mejores prácticas: Asegúrate que 'optimize for ad hoc workloads' esté habilitado (1) en entornos con muchas consultas ad-hoc.

Consultas de Rendimiento

Consultas Costosas

-- Top 20 consultas más costosas por CPU SELECT TOP 20 qs.execution_count, qs.total_logical_reads/qs.execution_count AS avg_logical_reads, qs.total_elapsed_time/qs.execution_count AS avg_elapsed_time, qs.total_worker_time/qs.execution_count AS avg_cpu_time, SUBSTRING(qt.text, (qs.statement_start_offset/2)+1, ((CASE qs.statement_end_offset WHEN -1 THEN DATALENGTH(qt.text) ELSE qs.statement_end_offset END - qs.statement_start_offset)/2)+1) AS query_text, qt.dbid, qt.objectid, qp.query_plan FROM sys.dm_exec_query_stats qs CROSS APPLY sys.dm_exec_sql_text(qs.sql_handle) qt CROSS APPLY sys.dm_exec_query_plan(qs.plan_handle) qp ORDER BY qs.total_worker_time/qs.execution_count DESC;

Índices No Utilizados

-- Índices no utilizados (lecturas = 0) SELECT OBJECT_NAME(i.object_id) AS TableName, i.name AS IndexName, i.type_desc AS IndexType, s.user_seeks, s.user_scans, s.user_lookups, s.user_updates FROM sys.indexes i INNER JOIN sys.objects o ON i.object_id = o.object_id LEFT JOIN sys.dm_db_index_usage_stats s ON s.object_id = i.object_id AND s.index_id = i.index_id AND s.database_id = DB_ID() WHERE o.type = 'U' -- Tablas de usuario AND i.name IS NOT NULL AND i.type_desc <> 'HEAP' AND (s.user_seeks = 0 OR s.user_seeks IS NULL) AND (s.user_scans = 0 OR s.user_scans IS NULL) AND (s.user_lookups = 0 OR s.user_lookups IS NULL) ORDER BY OBJECT_NAME(i.object_id), i.name;

Bloqueos y Deadlocks

-- Sesiones bloqueadas actualmente SELECT tl.request_session_id, wt.blocking_session_id, DB_NAME(tl.resource_database_id) AS DatabaseName, tl.resource_type, tl.resource_associated_entity_id, tl.request_mode, tl.request_status, st.text AS SQLText FROM sys.dm_tran_locks tl LEFT JOIN sys.dm_os_waiting_tasks wt ON tl.lock_owner_address = wt.resource_address LEFT JOIN sys.dm_exec_requests er ON tl.request_session_id = er.session_id OUTER APPLY sys.dm_exec_sql_text(er.sql_handle) st WHERE tl.resource_database_id = DB_ID() ORDER BY tl.request_session_id; -- Configurar trace de deadlocks (ejecutar en master) DBCC TRACEON (1222, -1);

Procedimientos Almacenados Esenciales

sp_WhoIsActive (Versión simplificada)

CREATE PROCEDURE dbo.sp_WhoIsActive AS BEGIN SELECT s.session_id, r.status, DB_NAME(r.database_id) AS database_name, s.login_name, s.host_name, r.command, r.wait_type, r.wait_time, r.wait_resource, r.cpu_time, r.logical_reads, r.reads, r.writes, t.text AS sql_text, qp.query_plan, r.start_time, r.percent_complete FROM sys.dm_exec_sessions s LEFT JOIN sys.dm_exec_requests r ON s.session_id = r.session_id OUTER APPLY sys.dm_exec_sql_text(r.sql_handle) t OUTER APPLY sys.dm_exec_query_plan(r.plan_handle) qp WHERE s.is_user_process = 1 ORDER BY r.cpu_time DESC; END;

Procedimiento de Mantenimiento de Índices

CREATE PROCEDURE dbo.IndexMaintenance @DatabaseName NVARCHAR(128), @FragmentationThreshold INT = 30 AS BEGIN DECLARE @SQL NVARCHAR(MAX); DECLARE @TableName NVARCHAR(128); DECLARE @IndexName NVARCHAR(128); DECLARE @SchemaName NVARCHAR(128); DECLARE @Fragmentation FLOAT; -- Crear tabla temporal para resultados CREATE TABLE #FragmentedIndexes ( TableName NVARCHAR(128), SchemaName NVARCHAR(128), IndexName NVARCHAR(128), Fragmentation FLOAT ); -- Consulta para encontrar índices fragmentados SET @SQL = N' INSERT INTO #FragmentedIndexes SELECT OBJECT_NAME(ind.object_id) AS TableName, SCHEMA_NAME(obj.schema_id) AS SchemaName, ind.name AS IndexName, ips.avg_fragmentation_in_percent AS Fragmentation FROM ' + QUOTENAME(@DatabaseName) + '.sys.dm_db_index_physical_stats(DB_ID(''' + @DatabaseName + '''), NULL, NULL, NULL, ''LIMITED'') ips JOIN ' + QUOTENAME(@DatabaseName) + '.sys.indexes ind ON ips.object_id = ind.object_id AND ips.index_id = ind.index_id JOIN ' + QUOTENAME(@DatabaseName) + '.sys.objects obj ON ind.object_id = obj.object_id WHERE ips.avg_fragmentation_in_percent > ' + CAST(@FragmentationThreshold AS NVARCHAR(10)) + ' AND ind.name IS NOT NULL AND obj.type = ''U'' ORDER BY ips.avg_fragmentation_in_percent DESC;'; EXEC sp_executesql @SQL; -- Procesar cada índice fragmentado DECLARE IndexCursor CURSOR FOR SELECT SchemaName, TableName, IndexName, Fragmentation FROM #FragmentedIndexes; OPEN IndexCursor; FETCH NEXT FROM IndexCursor INTO @SchemaName, @TableName, @IndexName, @Fragmentation; WHILE @@FETCH_STATUS = 0 BEGIN PRINT 'Procesando índice: ' + @SchemaName + '.' + @TableName + '.' + @IndexName + ' (Fragmentación: ' + CAST(@Fragmentation AS NVARCHAR(10)) + '%)'; IF @Fragmentation > 30 BEGIN -- Rebuild para fragmentación alta SET @SQL = 'ALTER INDEX ' + QUOTENAME(@IndexName) + ' ON ' + QUOTENAME(@DatabaseName) + '.' + QUOTENAME(@SchemaName) + '.' + QUOTENAME(@TableName) + ' REBUILD;'; EXEC sp_executesql @SQL; PRINT ' REBUILD ejecutado'; END ELSE BEGIN -- Reorganize para fragmentación moderada SET @SQL = 'ALTER INDEX ' + QUOTENAME(@IndexName) + ' ON ' + QUOTENAME(@DatabaseName) + '.' + QUOTENAME(@SchemaName) + '.' + QUOTENAME(@TableName) + ' REORGANIZE;'; EXEC sp_executesql @SQL; PRINT ' REORGANIZE ejecutado'; END FETCH NEXT FROM IndexCursor INTO @SchemaName, @TableName, @IndexName, @Fragmentation; END CLOSE IndexCursor; DEALLOCATE IndexCursor; DROP TABLE #FragmentedIndexes; END;

Scripts de Backup y Restauración

Backup Completo

-- Backup completo con compresión y verificación DECLARE @BackupPath NVARCHAR(255) = 'C:\Backups\' + DB_NAME() + '_' + REPLACE(REPLACE(REPLACE(CONVERT(VARCHAR, GETDATE(), 120), '-', ''), ' ', '_'), ':', '') + '.bak'; BACKUP DATABASE [YourDatabase] TO DISK = @BackupPath WITH COMPRESSION, STATS = 10, CHECKSUM, COPY_ONLY; -- Verificar el backup RESTORE VERIFYONLY FROM DISK = @BackupPath WITH CHECKSUM;

Backup Diferencial

-- Backup diferencial DECLARE @DiffBackupPath NVARCHAR(255) = 'C:\Backups\' + DB_NAME() + 'Diff' + REPLACE(REPLACE(REPLACE(CONVERT(VARCHAR, GETDATE(), 120), '-', ''), ' ', '_'), ':', '') + '.bak'; BACKUP DATABASE [YourDatabase] TO DISK = @DiffBackupPath WITH DIFFERENTIAL, COMPRESSION, STATS = 10, CHECKSUM;

Restauración de Base de Datos

-- Restaurar con reemplazo y recuperación USE [master]; GO ALTER DATABASE [YourDatabase] SET SINGLE_USER WITH ROLLBACK IMMEDIATE; GO RESTORE DATABASE [YourDatabase] FROM DISK = 'C:\Backups\YourDatabase_Full.bak' WITH FILE = 1, MOVE 'YourDatabase_Data' TO 'C:\Data\YourDatabase.mdf', MOVE 'YourDatabase_Log' TO 'C:\Data\YourDatabase.ldf', REPLACE, STATS = 10, RECOVERY; ALTER DATABASE [YourDatabase] SET MULTI_USER; GO

Flujo de Trabajo Básico

# Inicializar un nuevo repositorio mkdir dba-scripts cd dba-scripts git init # Crear estructura de directorios mkdir scripts backups documentation schemas # Crear archivo .gitignore para archivos no deseados echo "# Ignorar archivos de backup SQL" > .gitignore echo "*.bak" >> .gitignore echo "*.trn" >> .gitignore echo "*.ldf" >> .gitignore echo "*.mdf" >> .gitignore echo "*.ndf" >> .gitignore # Añadir y confirmar cambios git add . git commit -m "Estructura inicial del repositorio DBA" # Conectar con un repositorio remoto (GitHub, Azure DevOps, etc.) git remote add origin https://github.com/tuusuario/dba-scripts.git git push -u origin main

Comandos Avanzados Útiles

# Ver historial de cambios git log --oneline --graph --all # Crear una nueva rama para un cambio específico git checkout -b feature/index-optimization # Fusionar cambios a la rama principal git checkout main git merge feature/index-optimization # Guardar cambios temporalmente (stash) git stash save "Trabajo en progreso antes de cambiar de rama" # Recuperar cambios guardados git stash pop # Revertir cambios no confirmados git checkout -- nombre_archivo.sql # Crear un tag para una versión importante git tag -a v1.0.0 -m "Versión estable inicial" git push origin v1.0.0

Triggers y Funciones Útiles

Trigger de Auditoría para Cambios

CREATE TABLE dbo.AuditLog ( AuditLogID INT IDENTITY(1,1) PRIMARY KEY, TableName NVARCHAR(128), OperationType CHAR(1), -- 'I'=Insert, 'U'=Update, 'D'=Delete PrimaryKeyValue NVARCHAR(255), OldData NVARCHAR(MAX), NewData NVARCHAR(MAX), ChangedBy NVARCHAR(128), ChangeDate DATETIME DEFAULT GETDATE() ); CREATE TRIGGER tr_AuditCustomerChanges ON dbo.Customers AFTER INSERT, UPDATE, DELETE AS BEGIN SET NOCOUNT ON; -- Para INSERTS IF EXISTS (SELECT * FROM inserted) AND NOT EXISTS (SELECT * FROM deleted) BEGIN INSERT INTO dbo.AuditLog (TableName, OperationType, PrimaryKeyValue, NewData, ChangedBy) SELECT 'Customers', 'I', CAST(i.CustomerID AS NVARCHAR(255)), (SELECT * FROM inserted AS i FOR JSON AUTO), SYSTEM_USER FROM inserted i; END -- Para DELETES IF EXISTS (SELECT * FROM deleted) AND NOT EXISTS (SELECT * FROM inserted) BEGIN INSERT INTO dbo.AuditLog (TableName, OperationType, PrimaryKeyValue, OldData, ChangedBy) SELECT 'Customers', 'D', CAST(d.CustomerID AS NVARCHAR(255)), (SELECT * FROM deleted AS d FOR JSON AUTO), SYSTEM_USER FROM deleted d; END -- Para UPDATES IF EXISTS (SELECT * FROM inserted) AND EXISTS (SELECT * FROM deleted) BEGIN INSERT INTO dbo.AuditLog (TableName, OperationType, PrimaryKeyValue, OldData, NewData, ChangedBy) SELECT 'Customers', 'U', CAST(i.CustomerID AS NVARCHAR(255)), (SELECT * FROM deleted AS d WHERE d.CustomerID = i.CustomerID FOR JSON AUTO), (SELECT * FROM inserted AS i2 WHERE i2.CustomerID = i.CustomerID FOR JSON AUTO), SYSTEM_USER FROM inserted i; END END;

Función para Generar Scripts de Objetos

CREATE FUNCTION dbo.GenerateObjectScript (@ObjectName NVARCHAR(255), @ObjectType NVARCHAR(50)) RETURNS NVARCHAR(MAX) AS BEGIN DECLARE @Script NVARCHAR(MAX); IF @ObjectType = 'TABLE' BEGIN SELECT @Script = 'CREATE TABLE ' + QUOTENAME(SCHEMA_NAME(schema_id)) + '.' + QUOTENAME(name) + ' (' + CHAR(13) + CHAR(10) + STUFF(( SELECT ',' + CHAR(13) + CHAR(10) + ' ' + QUOTENAME(c.name) + ' ' + tp.name + CASE WHEN tp.name IN ('varchar', 'nvarchar', 'char', 'nchar') THEN '(' + CASE WHEN c.max_length = -1 THEN 'MAX' ELSE CAST(c.max_length AS VARCHAR) END + ')' WHEN tp.name IN ('decimal', 'numeric') THEN '(' + CAST(c.precision AS VARCHAR) + ',' + CAST(c.scale AS VARCHAR) + ')' ELSE '' END + ' ' + CASE WHEN c.is_nullable = 0 THEN 'NOT NULL' ELSE 'NULL' END FROM sys.columns c JOIN sys.types tp ON c.user_type_id = tp.user_type_id WHERE c.object_id = OBJECT_ID(@ObjectName) ORDER BY c.column_id FOR XML PATH(''), TYPE).value('.', 'NVARCHAR(MAX)'), 1, 2, '') + CHAR(13) + CHAR(10) + ');' FROM sys.tables WHERE object_id = OBJECT_ID(@ObjectName); END ELSE IF @ObjectType = 'PROCEDURE' BEGIN SELECT @Script = OBJECT_DEFINITION(OBJECT_ID(@ObjectName)); END RETURN @Script; END;
